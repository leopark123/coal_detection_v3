## 1. 系统概述

### 1.1 设计背景
翻车机积煤检测系统V3.0是一套专为煤炭翻车机房设计的工业视觉安全监控系统。系统通过实时图像分析检测格栅上的积煤情况，防止积煤堆积导致的翻车机卡死、设备损坏等安全事故。

### 1.2 设计原则
- **安全第一**: 宁可漏报，不可误报，避免误报导致生产中断
- **实时响应**: 端到端延迟<100ms，满足工业实时性要求
- **高可靠**: 7×24小时连续运行，故障率<0.1%
- **易维护**: 模块化设计，支持远程监控和维护
- **可扩展**: 支持多点位、多设备的扩展需求

### 1.3 关键技术指标

| 指标类别 | 指标项 | 技术要求 | 验证方法 |
|---------|-------|---------|----------|
| 性能指标 | 单帧处理延迟 | ≤ 80ms | 性能测试 |
| 性能指标 | 端到端延迟 | ≤ 150ms | 集成测试 |
| 性能指标 | 跳帧率 | ≤ 10% | 长期运行测试 |
| 可靠性指标 | 连续运行时间 | >720小时 | 稳定性测试 |
| 可靠性指标 | 故障恢复时间 | <30秒 | 故障注入测试 |
| 安全指标 | 误报率 | <0.5% | 现场验收测试 |
| 安全指标 | 漏报率 | <2% | 现场验收测试 |

**镜头计算**：
```
工作距离（WD）= 4000mm
目标尺寸（横向）= 3000mm
传感器尺寸（横向）= 7.4mm
焦距（f）= WD × 传感器尺寸 / 目标尺寸
        = 4000 × 7.4 / 3000 = 9.87mm
选择标准焦距：12mm（考虑安装误差和调节余量）
```

```python
class SafetyHeartbeat:
    def __init__(self):
        self.heartbeat_counter = 0
        self.last_heartbeat_time = time.time()
        self.heartbeat_interval = 0.5  # 500ms
        self.timeout_threshold = 1.5   # 1.5秒超时

    def send_heartbeat(self):
        """发送心跳信号"""
        self.heartbeat_counter = (self.heartbeat_counter + 1) % 65536
        self.last_heartbeat_time = time.time()

        # 写入PLC心跳点位
        plc.write_tag('Detection.VisionHeartbeat', self.heartbeat_counter)
        plc.write_tag('Detection.LastHeartbeatTime',
                     time.strftime('%Y-%m-%d %H:%M:%S'))

    def check_heartbeat_timeout(self):
        """检查心跳超时"""
        current_time = time.time()
        if current_time - self.last_heartbeat_time > self.timeout_threshold:
            # 心跳超时，触发故障安全模式
            self.trigger_safety_mode('HEARTBEAT_TIMEOUT')

    def trigger_safety_mode(self, fault_reason):
        """触发安全模式"""
        logger.critical(f"安全模式触发: {fault_reason}")
        plc.write_tag('Detection.SystemReady', False)
        plc.write_tag('Detection.FaultCode', FAULT_COMM_TIMEOUT)
        plc.write_tag('Detection.SafetyMode', True)
```

```python
class SecureDataTransmission:
    def __init__(self):
        self.encryption_key = self.generate_key()
        self.hmac_key = self.generate_hmac_key()

    def encrypt_plc_data(self, data):
        """加密PLC通信数据"""
        # AES-256加密
        cipher = AES.new(self.encryption_key, AES.MODE_GCM)
        ciphertext, tag = cipher.encrypt_and_digest(data.encode())

        # 计算HMAC确保完整性
        hmac_value = hmac.new(self.hmac_key, ciphertext, hashlib.sha256).hexdigest()

        return {
            'ciphertext': base64.b64encode(ciphertext).decode(),
            'nonce': base64.b64encode(cipher.nonce).decode(),
            'tag': base64.b64encode(tag).decode(),
            'hmac': hmac_value,
            'timestamp': time.time()
        }

    def decrypt_plc_data(self, encrypted_data):
        """解密PLC通信数据"""
        # 验证HMAC
        ciphertext = base64.b64decode(encrypted_data['ciphertext'])
        expected_hmac = hmac.new(self.hmac_key, ciphertext, hashlib.sha256).hexdigest()

        if not hmac.compare_digest(expected_hmac, encrypted_data['hmac']):
            raise ValueError("数据完整性验证失败")

        # AES解密
        cipher = AES.new(self.encryption_key, AES.MODE_GCM,
                        nonce=base64.b64decode(encrypted_data['nonce']))
        plaintext = cipher.decrypt_and_verify(ciphertext,
                                            base64.b64decode(encrypted_data['tag']))

        return plaintext.decode()
```

```python
class OptimizedDetectionPipeline:
    """优化的检测管线"""

    def __init__(self):
        # 预创建对象，避免运行时分配
        self.clahe = cv2.createCLAHE(clipLimit=3.0, tileGridSize=(8, 8))
        self.ecc_criteria = (cv2.TERM_CRITERIA_EPS | cv2.TERM_CRITERIA_COUNT, 5, 0.01)

        # 预加载参考图像（多分辨率）
        self.reference_full = cv2.imread('config/reference_3072x2048.jpg')
        self.reference_small = cv2.resize(self.reference_full, (320, 240))
        self.reference_gray = cv2.cvtColor(self.reference_small, cv2.COLOR_BGR2GRAY)

        # 预计算掩码和ROI
        self.grid_mask = self._create_grid_mask()
        self.coal_roi_mask = self._create_coal_roi_mask()

        # 初始化ECC变换矩阵
        self.warp_matrix = np.eye(2, 3, dtype=np.float32)

        # 性能计时器
        self.performance_timer = PerformanceTimer()

        # 预热算法（JIT编译、缓存预加载）
        self._warmup_algorithms()

    def process_frame(self, frame: np.ndarray, frame_id: int) -> DetectionResult:
        """
        优化的单帧处理管线
        目标: <80ms处理时间
        """
        with self.performance_timer.measure('total_processing'):

            # 1. 图像质量检查 (2-3ms)
            with self.performance_timer.measure('quality_check'):
                if not self._check_image_quality(frame):
                    return DetectionResult(fault_code=FAULT_QUALITY_FAIL)

            # 2. 快速ECC配准 (15-20ms)
            with self.performance_timer.measure('ecc_alignment'):
                aligned_frame = self._fast_ecc_align(frame)

            # 3. CLAHE增强 (5-10ms)
            with self.performance_timer.measure('image_enhancement'):
                enhanced_frame = self._enhance_image(aligned_frame)

            # 4. 并行检测 (20-30ms)
            with self.performance_timer.measure('parallel_detection'):
                grid_result, coal_result = self._parallel_detect(enhanced_frame)

            # 5. 综合判定 (<1ms)
            with self.performance_timer.measure('judgment'):
                final_result = self._judge_result(grid_result, coal_result)

            # 6. 多帧投票 (2-3ms)
            with self.performance_timer.measure('voting'):
                voted_result = self._multi_frame_voting(final_result)

        # 性能监控
        total_time = self.performance_timer.get_last_measurement('total_processing')
        if total_time > 80:  # 超过80ms告警
            logger.warning(f"处理超时: {total_time:.2f}ms, 帧号: {frame_id}")

        return voted_result

    def _fast_ecc_align(self, frame: np.ndarray) -> np.ndarray:
        """
        快速ECC配准算法
        优化: 小图计算变换矩阵，大图应用变换
        """
        # 降采样到320x240进行ECC计算
        frame_small = cv2.resize(frame, (320, 240))
        frame_gray = cv2.cvtColor(frame_small, cv2.COLOR_BGR2GRAY)

        try:
            # 在小图上计算ECC变换（快速）
            _, warp_matrix_small = cv2.findTransformECC(
                self.reference_gray, frame_gray,
                np.eye(2, 3, dtype=np.float32),
                cv2.MOTION_TRANSLATION,
                self.ecc_criteria
            )

            # 缩放变换矩阵到原图尺寸
            scale_x = frame.shape[1] / 320
            scale_y = frame.shape[0] / 240
            self.warp_matrix = warp_matrix_small.copy()
            self.warp_matrix[0, 2] *= scale_x
            self.warp_matrix[1, 2] *= scale_y

        except cv2.error:
            # ECC失败，使用上一次的变换矩阵
            logger.warning("ECC配准失败，使用历史变换矩阵")

        # 应用变换到原图
        return cv2.warpAffine(frame, self.warp_matrix,
                             (frame.shape[1], frame.shape[0]))

    def _parallel_detect(self, frame: np.ndarray) -> Tuple[GridResult, CoalResult]:
        """并行执行格栅检测和积煤检测"""
        with ThreadPoolExecutor(max_workers=2) as executor:
            # 提交两个并行任务
            grid_future = executor.submit(self._detect_grid_visibility, frame)
            coal_future = executor.submit(self._detect_coal_coverage, frame)

            # 等待结果（并行执行）
            grid_result = grid_future.result()
            coal_result = coal_future.result()

        return grid_result, coal_result
```

```python
class ResourceScheduler:

    def __init__(self):
        self.cpu_count = multiprocessing.cpu_count()
        self.gpu_available = self._check_gpu_availability()

        )

        # GPU任务队列
        if self.gpu_available:
            self.gpu_executor = ThreadPoolExecutor(max_workers=1)

        # 资源使用监控
        self.resource_monitor = ResourceMonitor()

    def schedule_detection_task(self, frame: np.ndarray) -> Future:
        """智能调度检测任务"""
        cpu_usage = self.resource_monitor.get_cpu_usage()
        gpu_usage = self.resource_monitor.get_gpu_usage() if self.gpu_available else 100

        # 根据资源使用情况选择执行器
        if self.gpu_available and gpu_usage < 80:
            # GPU负载较低，使用GPU加速
            return self.gpu_executor.submit(self._gpu_detect, frame)
        elif cpu_usage < 70:
            # CPU负载较低，使用CPU并行
            return self.cpu_executor.submit(self._cpu_detect, frame)
        else:
            # 资源紧张，使用简化算法
            return self.cpu_executor.submit(self._lite_detect, frame)

    def _gpu_detect(self, frame: np.ndarray) -> DetectionResult:
        """GPU加速检测"""
        # 使用CUDA加速的OpenCV操作
        gpu_frame = cv2.cuda_GpuMat()
        gpu_frame.upload(frame)

        # GPU上进行图像处理
        gpu_gray = cv2.cuda.cvtColor(gpu_frame, cv2.COLOR_BGR2GRAY)
        gpu_enhanced = cv2.cuda.equalizeHist(gpu_gray)

        # 下载结果到CPU进行后续处理
        enhanced = gpu_enhanced.download()

        return self._standard_detect(enhanced)

    def _cpu_detect(self, frame: np.ndarray) -> DetectionResult:
        """CPU并行检测"""
        # 标准CPU检测算法
        return self._standard_detect(frame)

    def _lite_detect(self, frame: np.ndarray) -> DetectionResult:

        # 降低图像分辨率，简化算法
        small_frame = cv2.resize(frame, (1024, 768))
```

```python
            pending_data = self._get_pending_sync_data()

            if pending_data:
                # 批量上传数据
                upload_result = await self.data_uploader.upload_batch(pending_data)

                if upload_result.success:
                    self._mark_data_synced(pending_data)
                    logger.info(f"成功同步{len(pending_data)}条检测数据")
                else:
                    logger.error(f"数据同步失败: {upload_result.error}")

        except Exception as e:
            logger.error(f"数据同步异常: {e}")

    async def check_model_updates(self):
        """检查并下载模型更新"""
        try:
            # 检查云端是否有新的模型版本
            latest_version = await self.api_client.get_latest_model_version()
            current_version = self._get_current_model_version()

            if latest_version > current_version:
                logger.info(f"发现新模型版本: {latest_version}")

                # 下载新模型
                model_data = await self.model_downloader.download_model(latest_version)

                # 验证模型完整性
                if self._validate_model(model_data):
                    # 热更新模型
                    await self._hot_update_model(model_data)
                    logger.info(f"模型更新完成: {latest_version}")
                else:
                    logger.error("模型验证失败，更新中止")

        except Exception as e:
            logger.error(f"模型更新异常: {e}")

    async def _hot_update_model(self, model_data: bytes):
        """热更新检测模型（不停机更新）"""
        # 1. 加载新模型到临时实例
        temp_detector = DetectorFactory.create_from_model(model_data)

        # 2. 验证新模型功能
        test_result = await self._test_model_performance(temp_detector)

        if test_result.passed:
            # 3. 原子替换检测器实例
            old_detector = self.detection_manager.get_current_detector()
            self.detection_manager.replace_detector(temp_detector)

            # 4. 清理旧模型资源
            old_detector.cleanup()

            logger.info("模型热更新成功")
        else:
            logger.error("新模型性能测试失败，回滚到原模型")
            temp_detector.cleanup()
```
